;; --- Fontset for CJK, symbols, block elements, and nerd-icons ---
;; Must run after frame creation for display-graphic-p to be true.
(defun my/setup-fontsets ()
  "Configure fontsets for CJK, symbols, and block elements."
  (when (display-graphic-p)
    ;; CJK — Noto Sans CJK JP
    (dolist (range '(japanese-jisx0208 katakana-jisx0201 kana han cjk-misc))
      (set-fontset-font t range (font-spec :family "Noto Sans CJK JP")))

    ;; Block elements (U+2580-U+259F) — the ASCII art characters ▄█░▀
    ;; Use Symbols Nerd Font which has complete coverage
    (set-fontset-font t '(#x2580 . #x259F)
                      (font-spec :family "Symbols Nerd Font") nil 'prepend)

    ;; Box drawing (U+2500-U+257F) — ═║╔╗╚╝─│
    (set-fontset-font t '(#x2500 . #x257F)
                      (font-spec :family "Symbols Nerd Font") nil 'prepend)

    ;; Geometric shapes (U+25A0-U+25FF) — ▪▫●○◆◇
    (set-fontset-font t '(#x25A0 . #x25FF)
                      (font-spec :family "Symbols Nerd Font") nil 'prepend)

    ;; Nerd Font private use area (U+E000-U+F8FF, U+F0000-U+FFFFF)
    (set-fontset-font t '(#xE000 . #xF8FF)
                      (font-spec :family "Symbols Nerd Font") nil 'prepend)
    (set-fontset-font t '(#xF0000 . #xFFFFF)
                      (font-spec :family "Symbols Nerd Font") nil 'prepend)

    ;; Emoji (if Noto Color Emoji installed — run: sudo emerge media-fonts/noto-emoji)
    (when (find-font (font-spec :family "Noto Color Emoji"))
      (set-fontset-font t 'emoji
                        (font-spec :family "Noto Color Emoji") nil 'prepend))

    ;; General symbol fallback
    (set-fontset-font t 'symbol
                      (font-spec :family "Symbols Nerd Font") nil 'append)

    ;; Final unicode fallback — CJK for anything Iosevka can't render
    (set-fontset-font t 'unicode
                      (font-spec :family "Noto Sans CJK JP") nil 'append)))

    (add-hook 'window-setup-hook #'my/setup-fontsets)))
